cmake_minimum_required(VERSION 3.20)
project(NexaMachine VERSION 2.11.0 LANGUAGES CXX)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(nexa src/main.cpp src/vm.cpp src/assembler.cpp src/config.cpp src/virtual_disk.cpp src/vm_manager.cpp src/executable.cpp src/vfs.cpp src/kernel.cpp src/shell.cpp src/memory_manager.cpp)
target_include_directories(nexa PRIVATE include)
target_link_libraries(nexa PRIVATE OpenSSL::Crypto)
if(MSVC)
 target_compile_options(nexa PRIVATE /W4 /permissive-)
else()
 target_compile_options(nexa PRIVATE -Wall -Wextra -Wpedantic)
endif()
option(NEXA_BUILD_GUI "Build the optional Qt6 desktop interface" ON)
if(NEXA_BUILD_GUI)
 find_package(Qt6 COMPONENTS Widgets QUIET)
 if(Qt6Widgets_FOUND)
  add_executable(nexa_gui src/gui.cpp src/vm.cpp src/assembler.cpp)
  target_include_directories(nexa_gui PRIVATE include)
  target_link_libraries(nexa_gui PRIVATE Qt6::Widgets)
 else()
  message(STATUS "Qt6 Widgets not found; skipping nexa_gui")
 endif()
endif()
include(CTest)
if(BUILD_TESTING)
 add_executable(nexa_tests tests/assembler_tests.cpp src/assembler.cpp src/vm.cpp)
 target_include_directories(nexa_tests PRIVATE include)
 add_test(NAME nexa_assembler_tests COMMAND nexa_tests)
 add_executable(nexa_feature_tests tests/feature_tests.cpp src/config.cpp src/virtual_disk.cpp)
 add_executable(nexa_kernel_tests tests/kernel_tests.cpp src/kernel.cpp src/shell.cpp src/executable.cpp src/vfs.cpp src/vm.cpp src/assembler.cpp src/memory_manager.cpp)
 target_include_directories(nexa_kernel_tests PRIVATE include)
 target_link_libraries(nexa_kernel_tests PRIVATE OpenSSL::Crypto)
 add_test(NAME nexa_kernel_tests COMMAND nexa_kernel_tests)
 add_executable(nexa_memory_tests tests/memory_tests.cpp src/memory_manager.cpp)
 target_include_directories(nexa_memory_tests PRIVATE include)
 add_test(NAME nexa_memory_tests COMMAND nexa_memory_tests)
 add_executable(nexa_os_tests tests/os_tests.cpp src/executable.cpp src/vfs.cpp src/vm.cpp src/assembler.cpp)
 target_include_directories(nexa_os_tests PRIVATE include)
 add_test(NAME nexa_os_tests COMMAND nexa_os_tests)
 target_include_directories(nexa_feature_tests PRIVATE include)
 add_test(NAME nexa_feature_tests COMMAND nexa_feature_tests)
endif()
