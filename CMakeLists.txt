cmake_minimum_required(VERSION 3.21)
project(NexaMachine VERSION 2.34.0 LANGUAGES CXX)
include(GNUInstallDirs)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Shared, portable VM engine and C ABI for desktop/mobile host frontends.
add_library(nexa_vm_core STATIC src/vm.cpp src/assembler.cpp src/session_manager.cpp src/c_api.cpp src/virtual_disk.cpp)
target_include_directories(nexa_vm_core PUBLIC include)
set_target_properties(nexa_vm_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
 target_compile_options(nexa_vm_core PRIVATE /W4 /permissive-)
else()
 target_compile_options(nexa_vm_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

option(NEXA_BUILD_HOST_TOOLS "Build CLI and OS simulator utilities" ON)
option(NEXA_BUILD_GUI "Build the optional Qt6 desktop interface" ON)
option(NEXA_REQUIRE_GUI "Fail configuration if Qt6 desktop GUI dependencies are unavailable" OFF)

if(NEXA_BUILD_HOST_TOOLS)
 find_package(OpenSSL REQUIRED COMPONENTS Crypto)
 add_executable(nexa src/main.cpp src/config.cpp src/virtual_disk.cpp src/vm_manager.cpp src/executable.cpp src/vfs.cpp src/kernel.cpp src/shell.cpp src/memory_manager.cpp src/platform.cpp src/workspace.cpp)
 target_include_directories(nexa PRIVATE include)
 target_link_libraries(nexa PRIVATE nexa_vm_core OpenSSL::Crypto)
 if(WIN32)
  install(TARGETS nexa
   RUNTIME_DEPENDENCY_SET nexa_cli_runtime_dependencies
   RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
  install(RUNTIME_DEPENDENCY_SET nexa_cli_runtime_dependencies
   RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR}
   PRE_EXCLUDE_REGEXES "api-ms-" "ext-ms-"
   POST_EXCLUDE_REGEXES ".*[Ss]ystem32/.*\\.dll")
 elseif(APPLE)
  # The macOS CLI uses OpenSSL for password hashing. Bundle non-system
  # dylibs and resolve them relative to the installed bin/ executable.
  set_target_properties(nexa PROPERTIES INSTALL_RPATH "@loader_path/../lib")
  install(TARGETS nexa
   RUNTIME_DEPENDENCY_SET nexa_cli_runtime_dependencies
   RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
  install(RUNTIME_DEPENDENCY_SET nexa_cli_runtime_dependencies
   LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
   PRE_EXCLUDE_REGEXES "^/System/Library/" "^/usr/lib/"
   POST_EXCLUDE_REGEXES "^/System/Library/" "^/usr/lib/")
 else()
  install(TARGETS nexa RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
 endif()
 if(MSVC)
  target_compile_options(nexa PRIVATE /W4 /permissive-)
 else()
  target_compile_options(nexa PRIVATE -Wall -Wextra -Wpedantic)
 endif()
 include(CTest)
 if(BUILD_TESTING)
  target_compile_definitions(nexa_vm_core PRIVATE NEXA_ENABLE_TEST_HOOKS)
  add_executable(nexa_tests tests/assembler_tests.cpp)
  target_link_libraries(nexa_tests PRIVATE nexa_vm_core)
  add_executable(nexa_platform_tests tests/platform_tests.cpp src/platform.cpp)

  target_include_directories(nexa_disk_scheduler_tests PRIVATE include)

  add_executable(nexa_session_tests tests/session_tests.cpp)
  target_link_libraries(nexa_session_tests PRIVATE nexa_vm_core)
  target_include_directories(nexa_session_tests PRIVATE include)
  add_test(NAME nexa_session_tests COMMAND nexa_session_tests)
  add_executable(nexa_snapshot_c_api_recovery_tests tests/snapshot_c_api_recovery_tests.cpp)
  target_link_libraries(nexa_snapshot_c_api_recovery_tests PRIVATE nexa_vm_core)
   target_compile_definitions(nexa_snapshot_c_api_recovery_tests PRIVATE NEXA_ENABLE_TEST_HOOKS)
  target_include_directories(nexa_snapshot_c_api_recovery_tests PRIVATE include)
  add_test(NAME nexa_snapshot_c_api_recovery_tests COMMAND nexa_snapshot_c_api_recovery_tests)
  add_executable(nexa_snapshot_journal_tests tests/snapshot_journal_tests.cpp)
  target_include_directories(nexa_snapshot_journal_tests PRIVATE include)
  add_test(NAME nexa_snapshot_journal_tests COMMAND nexa_snapshot_journal_tests)
  add_executable(nexa_c_api_tests tests/c_api_tests.cpp)
  target_link_libraries(nexa_c_api_tests PRIVATE nexa_vm_core)
  target_compile_definitions(nexa_c_api_tests PRIVATE NEXA_ENABLE_TEST_HOOKS)
  target_include_directories(nexa_c_api_tests PRIVATE include)
  add_test(NAME nexa_c_api_tests COMMAND nexa_c_api_tests)

 # Keep assertions active in Release test binaries; several legacy tests use assert
 # to execute API calls, and NDEBUG would silently remove those calls.
 if(BUILD_TESTING)
  foreach(test_target nexa_tests nexa_platform_tests nexa_disk_scheduler_tests nexa_session_tests nexa_snapshot_c_api_recovery_tests nexa_snapshot_journal_tests nexa_c_api_tests nexa_workspace_tests nexa_feature_tests nexa_kernel_tests nexa_memory_tests nexa_os_tests nexa_process_helper_tests)
   if(TARGET ${test_target})
    if(MSVC)
     target_compile_options(${test_target} PRIVATE /UNDEBUG)
    else()
     target_compile_options(${test_target} PRIVATE -UNDEBUG)
    endif()
   endif()
  endforeach()
 endif()

if(NEXA_BUILD_GUI)
 find_package(Qt6 COMPONENTS Widgets QUIET)
 if(Qt6Widgets_FOUND)
  set(NEXA_GUI_SOURCES
   src/gui.cpp src/session_gui.cpp src/config.cpp src/workspace.cpp src/virtual_disk.cpp)
  if(APPLE)
   # Qt's macOS deployment helper requires a proper .app bundle. A plain
   # executable skips framework/plugin deployment and crashes after packaging.
   add_executable(nexa_gui MACOSX_BUNDLE ${NEXA_GUI_SOURCES})
   set_target_properties(nexa_gui PROPERTIES
    MACOSX_BUNDLE_GUI_IDENTIFIER "org.nexa.machine.desktop"
    MACOSX_BUNDLE_BUNDLE_NAME "Nexa Machine"
    MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}")
   install(TARGETS nexa_gui BUNDLE DESTINATION ${CMAKE_INSTALL_BINDIR})
  else()
   add_executable(nexa_gui ${NEXA_GUI_SOURCES})
   install(TARGETS nexa_gui RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
  endif()
  target_include_directories(nexa_gui PRIVATE include)
  target_link_libraries(nexa_gui PRIVATE Qt6::Widgets nexa_vm_core)
  # Linux packages rely on distribution-provided Qt6 runtime dependencies:
  # the generic script can fail rewriting RPATH on distro plugin ELF files.
  if(WIN32 OR APPLE)
   qt_generate_deploy_app_script(TARGET nexa_gui OUTPUT_SCRIPT NEXA_QT_DEPLOY_SCRIPT NO_UNSUPPORTED_PLATFORM_ERROR)
   install(SCRIPT ${NEXA_QT_DEPLOY_SCRIPT})
  endif()
 elseif(NEXA_REQUIRE_GUI)
  message(FATAL_ERROR "NEXA_REQUIRE_GUI is ON but Qt6 Widgets was not found")
 else()
  message(STATUS "Qt6 Widgets not found; skipping nexa_gui")
 endif()
endif()
